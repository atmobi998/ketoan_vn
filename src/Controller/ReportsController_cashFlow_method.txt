    // Báo cáo lưu chuyển tiền tệ - B03-DN TT200
    public function cashFlow()
    {
        $session = $this->getRequest()->getSession();
        $periodTbl = $this->fetchTable('AccountingPeriods');
        $options = $periodTbl->find('list',['keyField' => 'code','valueField' => 'name'])->toArray();
        $this->set(compact('options'));

        $from_period = $this->request->getQuery('from_period');
        $to_period = $this->request->getQuery('to_period');
        $today = \Cake\I18n\FrozenDate::today();
        $curperiod = $periodTbl->find()->where(function (\Cake\Database\Expression\QueryExpression $exp, \Cake\ORM\Query $q) use ($today) {return $exp->lte('AccountingPeriods.start_date', $today)->gte('AccountingPeriods.end_date', $today);})->first();

        if (empty($from_period)) $from_period = $session->read('from_period') ?? ($curperiod->code ?? null);
        if (empty($to_period)) $to_period = $session->read('to_period') ?? ($curperiod->code ?? null);

        $session->write('from_period', $from_period);
        $session->write('to_period', $to_period);
        $this->set(compact('from_period','to_period'));

        $fromPeriod = $periodTbl->find()->where(['code' => $from_period])->first();
        $toPeriod = $periodTbl->find()->where(['code' => $to_period])->first();

        if (!$fromPeriod || !$toPeriod) {
            $this->Flash->error('Vui lòng chọn kỳ kế toán');
            return;
        }

        $service = new \App\Service\CashFlowService();
        $report = $service->getCashFlowReport($fromPeriod->start_date->format('Y-m-d'), $toPeriod->end_date->format('Y-m-d'));
        $detail = $service->getCashBalanceDetail($fromPeriod->start_date->format('Y-m-d'), $toPeriod->end_date->format('Y-m-d'));

        $this->set(compact('report','detail','fromPeriod','toPeriod'));

        if ($this->request->getQuery('export') === 'excel') {
            $excelService = new \App\Service\ExcelExportService();
            $file = $excelService->exportCashFlow($report, $detail);
            return $this->response->withFile($file, ['download' => true, 'name' => 'LCTT_TT200_'.$from_period.'_'.$to_period.'.xlsx']);
        }
        if ($this->request->getQuery('export') === 'pdf') {
            $pdfService = new \App\Service\PdfExportService();
            $file = $pdfService->exportCashFlow($report, $detail);
            return $this->response->withFile($file, ['download' => true, 'name' => 'LCTT_TT200_'.$from_period.'_'.$to_period.'.pdf']);
        }
    }
